/**
 * 动态规划
 * 时间复杂度：O(mn) 空间复杂度：O(mn)
 * 生成dp数组 填充为0
 * 从右下角遍历到左上角
 * 4种情况
 * - 右下角
 * - 最后一行
 * - 最后一列
 * - 非特殊情况
 * @param {*} grid
 * @returns
 */
var minPathSum = function (grid) {
  const m = grid.length
  const n = grid[0].length
  const dp = new Array(m).fill().map(() => new Array(n).fill(0))

  for (let i = m - 1; i >= 0; i--) {
    for (let j = n - 1; j >= 0; j--) {
      if (i == m - 1 && j == n - 1) {
        dp[i][j] = grid[i][j] //处理右下角
      } else if (i == m - 1 && j != n - 1) {
        dp[i][j] = grid[i][j] + dp[i][j + 1] // 最后一行 只能向前走j+1 但不是右下角
      } else if (i != m - 1 && j == n - 1) {
        dp[i][j] = grid[i][j] + dp[i + 1][j] // 最后一列 只能向下走i+1
      } else {
        dp[i][j] = grid[i][j] + Math.min(dp[i + 1][j], dp[i][j + 1]) // 非特殊情况
      }
    }
  }

  return dp[0][0]
}
